🔄 FLUXO DE DADOS
═════════════════════════════════════════════════════════════════════════════════

AppContext (aguaMl, metas.aguaMl, adicionarAgua)
    ↓
    ├→ Hoje.jsx
    │   └→ AnelHidratacao (props: consumidoMl, metaMl, onClickAdicionar)
    │       ├→ Calcula: pct = (consumidoMl / metaMl) * 100
    │       ├→ Determina: estado baseado em pct
    │       ├→ Renderiza: SVG + texto + botões
    │       └→ Evento: onClick → onClickAdicionar → atualiza aguaMl
    │
    └→ Re-renderiza com novos dados

💻 EXEMPLO DE USO
═════════════════════════════════════════════════════════════════════════════════

import AnelHidratacao from './AnelHidratacao.jsx'

export default function MinhaTelaDeHidratacao() {
  const [agua, setAgua] = useState(1500)  // ml
  const metaAgua = 2500  // ml

  function adicionarAgua(ml) {
    setAgua(Math.max(0, agua + ml))
  }

  return (
    <section className="rounded-2xl bg-white p-8">
      <h2 className="text-sm font-semibold text-verde/60">
        💧 Estratégia de Hidratação
      </h2>
      <p className="mb-6 text-xs text-verde/40">
        A hidratação estratégica sinaliza ao seu corpo que ele pode funcionar
        em plena capacidade
      </p>
      <AnelHidratacao
        consumidoMl={agua}
        metaMl={metaAgua}
        onClickAdicionar={adicionarAgua}
      />
    </section>
  )
}
